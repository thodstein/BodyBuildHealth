// Helper to map severity string to number
const severityToNumber = (sev: string): number => {
  switch (sev.toUpperCase()) {
    case "HIGH": return 3;
    case "MEDIUM": return 2;
    case "LOW": return 1;
    default: return 1;
  }
};

// Map InteractionType string to enum
const interactionTypeToEnum = (type: string): "synergy" | "conflict" | "danger" | "caution" => {
  switch (type.toUpperCase()) {
    case "SYNERGY": return "synergy";
    case "CONFLICT": return "conflict";
    case "DANGER": return "danger";
    default: return "caution";
  }
};
